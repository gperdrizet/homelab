    Users to Create¶

    -- LogKeep user
CREATE USER logkeep_user WITH PASSWORD '<secure_password>';
GRANT ALL PRIVILEGES ON DATABASE logkeep_prod TO logkeep_user;
GRANT ALL PRIVILEGES ON DATABASE logkeep_staging TO logkeep_user;
